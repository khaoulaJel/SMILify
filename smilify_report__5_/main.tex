\DeclareFinding{F1}
  {Chamfer distance can fall while correspondence is anatomically wrong; low
   loss is not evidence of a correct fit.}
  {Confirmed three times independently: rotation probe, 10-specimen fitter
   batch, synthetic ground truth.}
  {Established}

\DeclareFinding{F2}
  {Input mesh quality is not the dominant bottleneck. A carefully rebuilt
   preprocessing pipeline ties with the existing one on fit accuracy.}
  {Head-to-head Chamfer, 0.000771 vs.\ 0.000737.}
  {Established}

\DeclareFinding{F3}
  {Roughly 83\% of correspondence error never crosses a part boundary; it is
   sliding \emph{within} a correctly identified part.}
  {Synthetic ground-truth correspondence audit.}
  {Established}

\DeclareFinding{F4}
  {Initialisation, not the shape of the objective, dominates catastrophic
   registration failure.}
  {Ground-truth-initialised vs.\ zero-initialised fits; pose-sweep sensitivity.}
  {Established}

\DeclareFinding{F5}
  {Thin distal structures are systematically under-supported by sampling,
   independently of pose.}
  {Vertex-support audit: 38 tarsus / 8 pretarsus vertices vs.\ hundreds
   proximally.}
  {Established}
